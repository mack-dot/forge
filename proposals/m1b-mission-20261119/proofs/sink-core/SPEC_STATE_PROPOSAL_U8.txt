  // Stored state of one charge (owner-approved intent, if approved).
  datatype StoredCharge = StoredCharge(
    chargeId: string, original: int, remaining: int, refundEnabled: bool, settled: set<string>)

  // Where the inputs of Assess come from: the stored state. A request is open iff its id is not settled.
  function ContextAt(s: StoredCharge, requestId: string): Context {
    ChargeContext(s.chargeId, requestId, s.original, s.remaining, s.refundEnabled, requestId !in s.settled)
  }

  // The approved stored-state step. An authorization decreases remaining by the amount and settles the
  // request id. Every refusal changes nothing and has no intent.
  predicate Settles(pre: StoredCharge, requestId: string, response: ModelResponse,
                    post: StoredCharge, plan: seq<RefundIntent>) {
    if Eligible(ContextAt(pre, requestId), response) then
      post == pre.(remaining := pre.remaining - response.amount, settled := pre.settled + {requestId}) &&
      plan == [Intent(pre.chargeId, requestId, response.amount, response.reason)]
    else post == pre && plan == []
  }
